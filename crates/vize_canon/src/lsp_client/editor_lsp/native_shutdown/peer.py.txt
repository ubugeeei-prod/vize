#!/usr/bin/env python3
"""Owned protocol fixture: retain every whole frame and the actual stdin EOF."""
import json
import sys


def retain(packet):
    with open(LOG_PATH, "a", encoding="utf-8") as stream:
        stream.write(json.dumps(packet) + "\n")


while True:
    length = None
    while True:
        line = sys.stdin.buffer.readline()
        if not line:
            retain({"event": "stdin-eof"})
            sys.exit(0)
        if line == b"\r\n":
            break
        key, value = line.decode().strip().split(":", 1)
        if key.lower() == "content-length":
            length = int(value)
    request = json.loads(sys.stdin.buffer.read(length))
    retain({"direction": "request", "wholePacket": request})
    if request["method"] == "shutdown":
        response = {"jsonrpc": "2.0", "id": request["id"]}
        if REPLY_MODE == "error":
            response["error"] = {
                "code": -32603,
                "message": "controlled shutdown refusal",
                "data": {"phase": "shutdown"},
            }
        elif REPLY_MODE == "non-null":
            response["result"] = {"not": "null"}
        else:
            response["result"] = None
        retain({"direction": "response", "wholePacket": response})
        payload = json.dumps(response).encode()
        sys.stdout.buffer.write(f"Content-Length: {len(payload)}\r\n\r\n".encode() + payload)
        sys.stdout.buffer.flush()
