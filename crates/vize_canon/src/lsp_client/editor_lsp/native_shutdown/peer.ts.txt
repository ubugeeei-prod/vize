#!/usr/bin/env node
// Owned wire peer: retain every whole frame and actual stdin EOF.
import assert from "node:assert/strict";
import { appendFileSync } from "node:fs";

const logPath: string = LOG_PATH;
function options(mode: string | { shutdown: string; diagnostics: string }) {
  return typeof mode === "string" ? { shutdown: mode, diagnostics: "clean" } : mode;
}
const { shutdown, diagnostics } = options(REPLY_MODE);
type Packet = { jsonrpc: string; id?: number; method: string; params?: unknown };

function retain(packet: unknown) {
  appendFileSync(logPath, JSON.stringify(packet) + "\n");
}

async function respond(request: Packet, result: unknown, error?: unknown) {
  const response = error
    ? { jsonrpc: "2.0", id: request.id, error }
    : { jsonrpc: "2.0", id: request.id, result };
  retain({ direction: "response", wholePacket: response });
  const payload = Buffer.from(JSON.stringify(response));
  await new Promise<void>((resolve, reject) => {
    process.stdout.write(
      Buffer.concat([Buffer.from(`Content-Length: ${payload.length}\r\n\r\n`), payload]),
      (error) => error ? reject(error) : resolve(),
    );
  });
}

let pending = Buffer.alloc(0);
for await (const bytes of process.stdin) {
  pending = Buffer.concat([pending, bytes]);
  while (true) {
    const boundary = pending.indexOf("\r\n\r\n");
    if (boundary < 0) break;
    const header = pending.subarray(0, boundary).toString("ascii");
    const lengths = header.split("\r\n").filter((line) => /^content-length:/iu.test(line));
    assert.equal(lengths.length, 1);
    const length = Number(lengths[0].slice(lengths[0].indexOf(":") + 1));
    assert.ok(Number.isSafeInteger(length) && length >= 0);
    if (pending.length < boundary + 4 + length) break;
    const request: Packet = JSON.parse(pending.subarray(boundary + 4, boundary + 4 + length).toString());
    pending = pending.subarray(boundary + 4 + length);
    retain({ direction: "request", wholePacket: request });
    if (request.method === "initialize") {
      await respond(request, { capabilities: {} });
    } else if (request.method === "textDocument/documentSymbol") {
      await respond(request, []);
    } else if (request.method === "textDocument/diagnostic") {
      if (diagnostics === "error") {
        await respond(request, null, { code: -32603, message: "controlled diagnostic refusal", data: { phase: "diagnostics" } });
      } else {
        const items = diagnostics === "semantic" ? [{
          range: { start: { line: 0, character: 0 }, end: { line: 0, character: 1 } },
          severity: 1, code: 2322, message: "Type 'string' is not assignable to type 'number'.",
        }] : [];
        await respond(request, { kind: "full", items });
      }
    } else if (request.method === "shutdown") {
      if (shutdown === "error") {
        await respond(request, null, { code: -32603, message: "controlled shutdown refusal", data: { phase: "shutdown" } });
      } else {
        await respond(request, shutdown === "non-null" ? { not: "null" } : null);
      }
    } else {
      assert.equal(request.id, undefined, `unexpected request: ${request.method}`);
    }
  }
}
assert.equal(pending.length, 0, "partial frame at EOF");
retain({ event: "stdin-eof" });
