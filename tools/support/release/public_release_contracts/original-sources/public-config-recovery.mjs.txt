import path from 'node:path';

const env={...process.env,NODE_OPTIONS:'--stack-trace-limit=0'};
const CONFIG="export default async () => { await Promise.resolve(); throw new TypeError('owned-check-config-recovery'); };\n";
const CLEAN='<script setup lang="ts">\nconst n: number = 1;\n</script>\n<template><div>{{ n }}</div></template>\n';

const rows=[];let root;
for(const filename of ['vite.config.mjs','vite.config.ts']){root=fs.realpathSync(fs.mkdtempSync(path.join(tmpdir(),'vize-public-config-recovery-')));fs.writeFileSync(root+'/'+filename,CONFIG);fs.writeFileSync(root+'/A.vue',CLEAN);for(const explicit of [false,true]){const actual=packet(spawnSync(cli,['check','--format','json',...(explicit?['--config',filename]:[])],{cwd:root,env,encoding:'utf8',timeout:30000}));const before={status:2,signal:null,stdout:'',stderr:`\x1b[31mError:\x1b[0m failed to parse ${explicit?filename:root+'/'+filename}: node failed to load config: ${body}`};const expected={...before,stderr:before.stderr+HINT};const equal=(a,b)=>JSON.stringify(a)===JSON.stringify(b);rows.push({id:filename+(explicit?'-explicit':'-discovered'),directory:root,actual,expected,equal:equal(actual,expected),beforeEqual:equal(actual,before)});assert.equal(fs.readFileSync(root+'/'+filename,'utf8'),CONFIG);assert.equal(fs.readFileSync(root+'/A.vue','utf8'),CLEAN)}}

const result={workspace,node:process.version,oracle:packet(probe),rows};fs.writeFileSync(out,JSON.stringify(result,null,2)+'\n');console.log(JSON.stringify(rows.map(({id,equal,beforeEqual,error})=>({id,equal,beforeEqual,error}))));if(mode==='assert')assert.ok(rows.every(x=>x.equal));
