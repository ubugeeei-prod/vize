<script lang="ts">
import { Vue } from 'vue-class-component'
import { Component, PropSync } from 'vue-property-decorator'

@Component
export default class Panel extends Vue {
  @PropSync('open') synced!: boolean
  count = 0
  get doubled() {
    return this.count * 2
  }
}
</script>
<template><button @click="synced = 'yes'; doubled = 3">{{ count }}</button></template>
