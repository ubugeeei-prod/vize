<script setup lang="ts">
import type { ComponentPublicInstance } from "vue"
import Child from "./Child.vue"

function acceptElement(element: Element) {
  element.nodeName
}
</script>
<template>
  <Child :ref="
    (vnode: Element | ComponentPublicInstance | null) => {
      if (!vnode) return
      // @ts-ignore
      acceptElement(vnode.$el)
    }
  " />
</template>
