defineProps<{ static: boolean; class: boolean; default: string }>();

const count = 1;
const check = (value: {
  static: boolean; class: boolean; named: boolean; key: true;
  member: "member"; literal: "static"; template: "class";
  regex: boolean; ordinary: number;
}) => value.static;
</script>
<template>
  <button :disabled="static">{{ check({ static, class, named: default.length > 0, key: true, member: item.default, literal: 'static', template: `class`, regex: /static/.test(default), ordinary: count }) }}</button>
</template>
