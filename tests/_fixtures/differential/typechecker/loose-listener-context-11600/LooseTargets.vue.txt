type Component = new (...args: any[]) => { $emit: (...args: any[]) => void }
interface VNode { readonly __v_isVNode: true }
declare const target: Component
declare const node: VNode
declare const loose: any
declare const tag: string
const take = (value: unknown) => value
const forward = (value: string) => value
const forwardMany = (...values: unknown[]) => values
function generic<T>(value: T): T { return value }
function overloaded(value: string): string
function overloaded(value: number): number
function overloaded(value: string | number) { return value }
</script>

<component :is="node" @pick="forward($event); $event.foo; forwardMany($event)" />

<component :is="loose" @pick="overloaded" />
