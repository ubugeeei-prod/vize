<script setup lang="ts">
import { Dialog } from "./dialogs";
import { RegistrationType, type ParsedContentsInfoRaw, type ParsedQuestionItemRaw } from "./dialogs/types";

let isClosed = false;
let opened = true;

function onParsed({
  contentsInfo,
  questionItems,
}: {
  contentsInfo: ParsedContentsInfoRaw<RegistrationType.NormalOneto>;
  questionItems: ParsedQuestionItemRaw<RegistrationType.NormalOneto>[];
}) {
  contentsInfo.normalId.toUpperCase();
  questionItems[0]?.normalQuestionId.toUpperCase();
}

function cancel() {}
</script>

<template>
  <Dialog
    v-if="!isClosed"
    v-model="opened"
    :registration-type="RegistrationType.NormalOneto"
    @parsed="onParsed"
    @cancel="cancel"
  />
</template>
