    modelValue: { type: StringConstructor; default: string };
    rules: { type: PropType<ValidationRule<string>[]>; default: () => any; required: false };
    format: { type: PropType<{ display: (v: string) => string }>; required: false };
  },
  {},
  {},
  {},
  {},
  {},
  {},
  {
    "update:modelValue": (value: string) => true;
    clickAppendInner: (event: MouseEvent) => true;
  }
>;
