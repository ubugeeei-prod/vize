<script setup lang="ts">
import { ref } from "vue";

interface FooProps {
  kind?: "one";
  value?: string;
}

interface BarProps {
  kind?: "many";
  value?: string[];
}

type ExampleProps = FooProps | BarProps;

const props = withDefaults(defineProps<ExampleProps>(), {
  kind: "one",
});

const value = ref(props.value);

function clear() {
  value.value = undefined;
}
</script>

<template>
  <button type="button" @click="clear">Clear</button>
</template>
