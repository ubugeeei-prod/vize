<script setup lang="ts">
function useFlag(flag: boolean | (() => boolean)) {
  return flag
}

const props = withDefaults(defineProps<{ modal?: boolean; label?: string }>(), {
  modal: false,
  label: 'Dialog',
})
useFlag(() => props.label)
</script>

<template>
  <div :aria-label="props.label" />
</template>
