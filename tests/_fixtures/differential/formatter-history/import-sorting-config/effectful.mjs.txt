import { readFileSync, writeFileSync } from 'node:fs';
const counter = new URL('./eval-count.txt', import.meta.url);
export default () => {
  const count = Number(readFileSync(counter, 'utf8')) + 1;
  writeFileSync(counter, String(count));
  const config = { formatter: { sortImports: count === 1 ? {} : false },
    entries: [{ basePath: 'nested', files: ['*.vue'], ignores: ['Ignored.vue'] }] };
  writeFileSync(new URL('./returned-config.json', import.meta.url), JSON.stringify(config));
  return config;
};
