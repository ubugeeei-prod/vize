  _push(`<div${_ssrRenderAttrs(_attrs)}>${_ssrInterpolate($setup.describe("x"))} ${_ssrInterpolate($setup.count)}</div>`);
}
import { helper } from "./helper";
import { ref } from "vue";
export function describe(value) {
  return helper(value);
}
export default {
  __name: "Layout",
  ssrRender,
  setup(__props) {
    const count = ref(0);
    const __returned__ = {
      ref,
      count,

    Object.defineProperty(__returned__, "__isScriptSetup", {
      enumerable: false,
      value: true
    });
    return __returned__;
  }
};
