import { ssrInterpolate as _ssrInterpolate, ssrRenderAttrs as _ssrRenderAttrs } from "@vue/server-renderer"


function ssrRender(_ctx, _push, _parent, _attrs, $props, $setup, $data, $options) {
  _push(`<p${_ssrRenderAttrs(_attrs)}>${_ssrInterpolate($setup.tab)}${_ssrInterpolate($setup.raw)}${_ssrInterpolate($setup.tpl)}${_ssrInterpolate($setup.re)}${_ssrInterpolate($setup.pad())}</p>`)
}

const tab = "	";
const raw = "a	b";

export default {
  __name: 'Layout',
  ssrRender: ssrRender,
  setup(__props, { expose: __expose }) {
  __expose();

const tpl = `x	y
	z`;
const re = /	/;
/* keep	a comment tab */
function pad() {
  return `
	${tab}`;
}

const __returned__ = { re, pad, raw, tab, tpl }
Object.defineProperty(__returned__, '__isScriptSetup', { enumerable: false, value: true })
return __returned__
}

}
