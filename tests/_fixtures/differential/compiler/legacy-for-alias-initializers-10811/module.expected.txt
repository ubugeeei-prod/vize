import { ssrInterpolate as _ssrInterpolate, ssrRenderList as _ssrRenderList } from "@vue/server-renderer"

function ssrRender(_ctx, _push, _parent, _attrs) {
  _push(`<!--[-->`)
  _push(`<!--[-->`)
  _ssrRenderList(_ctx.rows, (outer) => {
    _push(`<div>`)
    _push(`<!--[-->`)
    _ssrRenderList(outer.items, ({ label = _ctx.fallback, [_ctx.key]: tag }) => {
      _push(`<p>${_ssrInterpolate(label + tag + _ctx.outside)}</p>`)
    })
    _push(`<!--]-->`)
    _push(`</div>`)
  })
  _push(`<!--]-->`)
  _push(`<!--]-->`)
}

const _sfc_main = {}
_sfc_main.ssrRender = ssrRender
export default _sfc_main
