import { ssrInterpolate as _ssrInterpolate, ssrRenderSlot as _ssrRenderSlot } from "@vue/server-renderer"

function ssrRender(_ctx, _push, _parent, _attrs) {
  _ssrRenderSlot(_ctx.$slots, "default", {}, () => {
    _push(`hello ${_ssrInterpolate(_ctx.msg)}`)
  }, _push, _parent)
}

const _sfc_main = {}
_sfc_main.ssrRender = ssrRender
export default _sfc_main
