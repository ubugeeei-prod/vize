<script setup>
import { ref } from 'vue'
const groups = ref([['a', 'b'], ['c']])
const itemElements = ref([])
const anchor = ref(null)
function add() { groups.value[1].push('d') }
function clear() { groups.value = [] }
function referencedText() { return itemElements.value.map((node) => node.textContent).sort().join(',') + '|' + (anchor.value?.textContent ?? 'null') }
defineExpose({ add, clear, referencedText })
</script>
<template><main><template v-for="(group, i) in groups" :key="i"><span v-for="item in group" :key="item" ref="itemElements">{{ item }}</span></template><aside ref="anchor">single</aside></main></template>
