function ssrRender(_ctx, _push, _parent, _attrs, $props, $setup, $data, $options) {
  _push(`<!--[-->`)
  if ($setup.ok) {
    _push(`<div>x</div>`)
  } else {
    _push(`<!---->`)
  }
  _push(`<!--[-->`)
  _ssrRenderList(3, (i) => {
    _push(`<div>a</div>`)
  })

  _push(`<!--]-->`)
}
