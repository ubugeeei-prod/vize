import { ssrRenderAttr as _ssrRenderAttr, ssrRenderList as _ssrRenderList } from "@vue/server-renderer"


function ssrRender(_ctx, _push, _parent, _attrs, $props, $setup, $data, $options) {
  _push(`<!--[--><div><div><Child :x="1">{{ x }}</Child><slot name="s">f</slot></div></div>`)
  _push(`<!--[-->`)
  _ssrRenderList($setup.rows, (row, { k }) => {
    _push(`<div><input${_ssrRenderAttr("value", k.x)}></div>`)
  })
  _push(`<!--]-->`)
  _push(`<!--]-->`)
}

export default {
  __name: 'Layout',
  ssrRender: ssrRender,
  setup(__props, { expose: __expose }) {
  __expose();

const rows = { a: { k: { x: 1 } } };

const __returned__ = { rows }
Object.defineProperty(__returned__, '__isScriptSetup', { enumerable: false, value: true })
return __returned__
}

}
