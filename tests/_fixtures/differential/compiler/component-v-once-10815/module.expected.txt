import { ssrInterpolate as _ssrInterpolate, ssrRenderComponent as _ssrRenderComponent } from "@vue/server-renderer"
import { createTextVNode as _createTextVNode, toDisplayString as _toDisplayString, withCtx as _withCtx } from "vue"


function ssrRender(_ctx, _push, _parent, _attrs, $props, $setup, $data, $options) {
  _push(`<!--[-->`)
  _push(_ssrRenderComponent($setup.MyButton, { label: $setup.label, onPing: $setup.onPing }, {
    default: _withCtx((_, _push, _parent, _scopeId) => {
      if (_push) {
        _push(`text ${_ssrInterpolate($setup.label)}`)
      } else {
        return [_createTextVNode("text "), _createTextVNode(_toDisplayString($setup.label))]
      }
    }),
    _: 1
  }, _parent))
  _push(_ssrRenderComponent($setup.MyButton, { label: "static" }, null, _parent))
  _push(`<!--]-->`)
}
import MyButton from './MyButton.vue'
import { useX } from './x'


export default {
  __name: 'Layout',
  ssrRender: ssrRender,
  setup(__props) {

const { label, onPing } = useX()

const __returned__ = { MyButton, useX, label, onPing }
Object.defineProperty(__returned__, '__isScriptSetup', { enumerable: false, value: true })
return __returned__
}

}
