<script setup>
import { computed, onUnmounted, ref } from 'vue';
import Child from './Child.vue';
import { record } from './fixture';

const count = ref(0);
const inRoom = computed(() => count.value >= 0);
const label = computed(() => `room-${count.value}`);
const increment = () => { count.value++; record(count.value); };
onUnmounted(() => record('parent-unmounted'));
</script>

<template>
  <section v-if="inRoom">
    <button id="increment" @click="increment">{{ count }}</button>
    <Child :label="label" />
  </section>
</template>
