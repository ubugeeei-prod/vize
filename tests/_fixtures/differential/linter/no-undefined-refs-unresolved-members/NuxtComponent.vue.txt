<script lang="ts">
export default defineNuxtComponent({
  props: ['title'],
  async asyncData() {
    const posts = await $fetch('/api/posts')
    return { posts }
  },
  data() {
    return { open: false }
  },
  methods: {
    toggle() {
      this.open = !this.open
    },
  },
})
</script>

<template>
  <h1 @click="toggle">{{ title }} {{ titel }}</h1>
  <ul v-if="open">
    <li v-for="post in posts" :key="post.id">{{ post.title }}</li>
  </ul>
</template>
