<script setup lang="ts">
import { nextTick, ref, watch } from "vue";

const props = defineProps<{ open: boolean; query: string }>();
const emit = defineEmits<{ opened: [] }>();
const editing = ref(false);
const lastQuery = ref("");

function refresh() {}

// side effects + resetting state when an event happens
watch(
  () => props.open,
  (open) => {
    editing.value = false;
    refresh();
    if (open) emit("opened");
  },
);

// keep the previous value unless a condition holds (not derivable with computed)
watch(
  () => props.query,
  (next) => {
    if (next.length >= 3) {
      lastQuery.value = next;
      nextTick(() => refresh());
    }
  },
);
</script>

<template>
  <label>
    Name <span class="required"> * </span>
    <input v-if="editing" />
  </label>
  <p>{{ lastQuery }}</p>
</template>
