<script setup lang="ts">
const pages: string[][] = [["a", "b"], ["c"]];
const current = 0;
function pick(index: number): string[] {
  return pages[index] ?? [];
}
</script>

<template>
  <ul>
    <li
      v-for="item in pages[
        current
      ]"
      :key="item"
    >{{ item }}</li>
    <li
      v-for="item in pick(
        current
      )"
      :key="item"
    >{{ item }}</li>
    <li
      v-for="item in
        pick(current)"
      :key="item"
    >{{ item }}</li>
    <li v-for="item in pick(current)" :key="item">{{ item }}</li>
  </ul>
</template>
