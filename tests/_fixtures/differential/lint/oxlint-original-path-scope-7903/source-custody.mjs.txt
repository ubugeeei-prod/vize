export default {
  meta: { name: "source-custody" },
  rules: {
    "original-path": {
      meta: { type: "problem", schema: [{ type: "object", properties: { marker: { type: "string" } }, additionalProperties: false }] },
      create(context) {
        return {
          Program(node) {
            context.report({ node, message: `Original script owner: ${context.getFilename()}; config: ${context.options[0].marker}` });
          },
        };
      },
    },
  },
};
